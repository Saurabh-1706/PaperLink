# Handwriting recogniser (LINE_RECOGNIZER=trocr). Deliberately NOT in
# requirements.txt: these are only needed when the flag is on, and torch alone is a
# multi-GB install that the CPU deployment has no use for.
#
# The GPU worker image already ships torch + CUDA (see docker/Dockerfile.worker.gpu),
# so torch is pinned only as a floor here, not re-downloaded.
#
# sentencepiece is not optional: without it, transformers misreads
# trocr-small-handwritten's tokenizer as a tiktoken file and model load fails with
# "`tiktoken` is required to read a `tiktoken` file".
torch>=2.4
transformers>=4.44
sentencepiece>=0.2
